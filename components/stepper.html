Usage

<div data-controller="stepper" 
     data-stepper-active-class-value="active" 
     data-stepper-completed-class-value="completed">
  
  <!-- Steps -->
  <div data-stepper-target="step" data-action="click->stepper#goTo" data-step="0">Step 1</div>
  <div data-stepper-target="step" data-action="click->stepper#goTo" data-step="1">Step 2</div>

  <!-- Contents -->
  <div data-stepper-target="content">Content 1</div>
  <div data-stepper-target="content" class="hidden">Content 2</div>

  <!-- Actions -->
  <button data-action="stepper#previous">Back</button>
  <button data-action="stepper#next">Next</button>
</div>
